### 3.2 샘플 데이터 생성 (실제 데이터 구조 기반)

실제 데이터가 없을 경우, FAOSTAT 구조를 기반으로 샘플 데이터를 생성합니다:

```python
def create_faostat_crops_sample():
    """
    FAOSTAT Crops Production 데이터 구조 기반 샘플 생성
    
    출처: FAO. 2024. FAOSTAT: Crops and livestock products. 
          https://www.fao.org/faostat/en/#data/QCL
    """
    np.random.seed(42)
    years = list(range(2000, 2024))
    
    # 분석 대상 품목 및 주요 생산국
    items_countries = {
        'Wheat': {
            'countries': ['China', 'India', 'Russian Federation', 'United States of America', 'France', 'Australia'],
            'base_production': [130, 100, 70, 50, 35, 25]  # 백만 톤
        },
        'Tomatoes': {
            'countries': ['China', 'India', 'Turkey', 'United States of America', 'Egypt', 'Italy'],
            'base_production': [60, 20, 12, 10, 8, 6]
        },
        'Lettuce and chicory': {
            'countries': ['China', 'United States of America', 'India', 'Spain', 'Italy', 'Japan'],
            'base_production': [15, 4, 1.2, 1, 0.8, 0.6]
        },
        'Potatoes': {
            'countries': ['China', 'India', 'Russian Federation', 'United States of America', 'Germany', 'Bangladesh'],
            'base_production': [90, 50, 30, 20, 10, 9]
        },
        'Chillies and peppers, dry': {
            'countries': ['India', 'China', 'Bangladesh', 'Thailand', 'Pakistan', 'Ethiopia'],
            'base_production': [2, 0.3, 0.2, 0.15, 0.14, 0.12]
        }
    }
    
    records = []
    
    for item, info in items_countries.items():
        for country, base_prod in zip(info['countries'], info['base_production']):
            for year in years:
                # 연도별 트렌드 (연간 2% 성장) + 기후 변동성
                trend = (year - 2000) * 0.02
                climate_shock = np.random.normal(0, 0.05)
                
                # 주요 기후 이벤트 반영
                if year == 2012 and country in ['United States of America', 'Russian Federation']:
                    climate_shock = -0.15  # 2012 가뭄
                if year == 2022 and item == 'Wheat' and country == 'Russian Federation':
                    climate_shock = -0.10  # 우크라이나 전쟁
                
                production = base_prod * (1 + trend + climate_shock) * 1_000_000
                area = production / (np.random.uniform(2.5, 4.5) * 10000)
                yield_val = (production / area) * 10 if area > 0 else 0  # hg/ha
                
                records.extend([
                    {'Area': country, 'Item': item, 'Element': 'Production', 
                     'Year': year, 'Unit': 'tonnes', 'Value': int(production)},
                    {'Area': country, 'Item': item, 'Element': 'Area harvested', 
                     'Year': year, 'Unit': 'ha', 'Value': int(area)},
                    {'Area': country, 'Item': item, 'Element': 'Yield', 
                     'Year': year, 'Unit': 'hg/ha', 'Value': int(yield_val)}
                ])
    
    return pd.DataFrame(records)


def create_faostat_livestock_sample():
    """
    FAOSTAT Livestock 데이터 구조 기반 샘플 생성
    
    출처: FAO. 2024. FAOSTAT: Crops and livestock products.
          https://www.fao.org/faostat/en/#data/QCL
    """
    np.random.seed(43)
    years = list(range(2000, 2024))
    
    countries_base = {
        'United States of America': 12,
        'Brazil': 10,
        'China': 7,
        'Argentina': 3,
        'Australia': 2,
        'India': 4
    }
    
    records = []
    for country, base in countries_base.items():
        for year in years:
            trend = (year - 2000) * 0.015
            shock = np.random.normal(0, 0.03)
            
            # 폭염 영향
            if year in [2011, 2012, 2021] and country == 'United States of America':
                shock = -0.08
            
            production = base * (1 + trend + shock) * 1_000_000
            records.append({
                'Area': country,
                'Item': 'Meat of cattle with the bone, fresh or chilled',
                'Element': 'Production',
                'Year': year,
                'Unit': 'tonnes',
                'Value': int(production)
            })
    
    return pd.DataFrame(records)


def create_faostat_temperature_sample():
    """
    FAOSTAT Temperature Change 데이터 구조 기반 샘플 생성
    
    출처: FAO. 2024. FAOSTAT: Temperature change on land.
          https://www.fao.org/faostat/en/#data/ET
    원본: NASA Goddard Institute for Space Studies (GISTEMP)
    """
    np.random.seed(44)
    years = list(range(1990, 2024))
    
    countries = ['World', 'China', 'India', 'United States of America', 
                 'Russian Federation', 'Brazil', 'Australia', 'Korea, Republic of']
    
    # 지역별 온난화 계수 (고위도 지역이 더 빠르게 온난화)
    regional_factors = {
        'World': 1.0,
        'Russian Federation': 1.4,
        'United States of America': 1.2,
        'China': 1.1,
        'Korea, Republic of': 1.1,
        'India': 0.9,
        'Brazil': 0.85,
        'Australia': 1.0
    }
    
    records = []
    for country in countries:
        factor = regional_factors.get(country, 1.0)
        for year in years:
            # 기준: 1951-1980 대비 기온 편차
            base_anomaly = (year - 1990) * 0.025
            variation = np.random.normal(0, 0.12)
            
            temp_change = base_anomaly * factor + variation
            
            # 최근 급격한 상승 반영
            if year >= 2015:
                temp_change += 0.2
            if year >= 2020:
                temp_change += 0.15
            
            records.append({
                'Area': country,
                'Months': 'Meteorological year',
                'Element': 'Temperature change',
                'Year': year,
                'Unit': '°C',
                'Value': round(temp_change, 3)
            })
    
    return pd.DataFrame(records)


def create_korea_trade_sample():
    """
    FAOSTAT Trade 데이터 구조 기반 한국 수입 샘플 생성
    
    출처: FAO. 2024. FAOSTAT: Crops and livestock products (Trade).
          https://www.fao.org/faostat/en/#data/TCL
    """
    np.random.seed(45)
    years = list(range(2000, 2024))
    
    items_base = {
        'Wheat': {'qty': 3_500_000, 'value': 800_000},  # 톤, 천달러
        'Potatoes': {'qty': 50_000, 'value': 30_000},
        'Meat of cattle, boneless (beef & veal)': {'qty': 400_000, 'value': 2_000_000},
        'Tomatoes': {'qty': 5_000, 'value': 10_000}
    }
    
    records = []
    for item, base in items_base.items():
        for year in years:
            trend = (year - 2000) * 0.03
            variation = np.random.normal(0, 0.08)
            
            qty = base['qty'] * (1 + trend + variation)
            value = base['value'] * (1 + trend + variation * 1.5)
            
            records.extend([
                {'Reporter Countries': 'Korea, Republic of', 'Partner Countries': 'World',
                 'Item': item, 'Element': 'Import Quantity', 'Year': year, 
                 'Unit': 'tonnes', 'Value': int(qty)},
                {'Reporter Countries': 'Korea, Republic of', 'Partner Countries': 'World',
                 'Item': item, 'Element': 'Import Value', 'Year': year, 
                 'Unit': '1000 US$', 'Value': int(value)}
            ])
    
    return pd.DataFrame(records)


def create_kaggle_climate_agriculture_sample():
    """
    Kaggle Climate Change Impact on Agriculture 데이터 구조 기반 샘플 생성
    
    출처: World Air Quality Index Project. 2024. 
          Climate Change Impact on Agriculture. Kaggle.
          https://www.kaggle.com/datasets/waqi786/climate-change-impact-on-agriculture
    """
    np.random.seed(46)
    years = list(range(1990, 2024))
    
    countries_regions = [
        ('United States of America', 'North America'),
        ('China', 'Asia'),
        ('India', 'Asia'),
        ('Brazil', 'South America'),
        ('Australia', 'Oceania'),
        ('France', 'Europe'),
        ('Russian Federation', 'Europe'),
        ('Korea, Republic of', 'Asia')
    ]
    
    crop_types = ['Wheat', 'Corn', 'Rice', 'Soybeans', 'Potatoes']
    
    records = []
    for country, region in countries_regions:
        for year in years:
            for crop in crop_types:
                # 기후 변수
                avg_temp = 14 + (year - 1990) * 0.03 + np.random.normal(0, 1)
                precipitation = np.random.uniform(400, 1200)
                co2 = 350 + (year - 1990) * 2.5 + np.random.normal(0, 5)  # ppm 기반
                
                # 수율 (기후 영향 반영)
                base_yield = np.random.uniform(2.5, 5.5)
                temp_effect = -0.05 * max(0, avg_temp - 16)
                precip_effect = 0.001 * (precipitation - 700)
                crop_yield = max(0.5, base_yield * (1 + temp_effect + precip_effect + np.random.normal(0, 0.1)))
                
                records.append({
                    'Year': year,
                    'Country': country,
                    'Region': region,
                    'Crop_Type': crop,
                    'Average_Temperature_C': round(avg_temp, 2),
                    'Total_Precipitation_mm': round(precipitation, 1),
                    'CO2_Emissions_MT': round(co2 * 10, 1),  # 국가 배출량 스케일
                    'Crop_Yield_MT_per_HA': round(crop_yield, 3),
                    'Pesticide_Use_KG_per_HA': round(np.random.uniform(1, 5), 2),
                    'Fertilizer_Use_KG_per_HA': round(np.random.uniform(100, 300), 1),
                    'Economic_Impact_Million_USD': round(crop_yield * np.random.uniform(50, 150), 2)
                })
    
    return pd.DataFrame(records)


# 샘플 데이터 생성 및 저장
print("📊 샘플 데이터 생성 중...")

df_livestock = create_faostat_livestock_sample()
df_temperature = create_faostat_temperature_sample()
df_trade = create_korea_trade_sample()
df_kaggle = create_kaggle_climate_agriculture_sample()

# 저장
df_crops.to_csv(RAW_DIR / 'faostat_crops.csv', index=False)
df_livestock.to_csv(RAW_DIR / 'faostat_livestock.csv', index=False)
df_temperature.to_csv(RAW_DIR / 'faostat_temperature.csv', index=False)
df_trade.to_csv(RAW_DIR / 'faostat_trade_korea.csv', index=False)
df_kaggle.to_csv(RAW_DIR / 'kaggle_climate_agriculture.csv', index=False)

print(f"✅ 샘플 데이터 생성 완료")
print(f"   - 작물 생산량: {len(df_crops):,} rows")
print(f"   - 축산 생산량: {len(df_livestock):,} rows")
print(f"   - 기온 변화: {len(df_temperature):,} rows")
print(f"   - 한국 수입: {len(df_trade):,} rows")
print(f"   - Kaggle 통합: {len(df_kaggle):,} rows")
```

```python
def preprocess_faostat_production(df, item_name, element='Production'):
    """
    FAOSTAT 생산량 데이터를 Wide format으로 변환
    
    Parameters:
    -----------
    df : DataFrame - 원본 FAOSTAT 데이터
    item_name : str - 품목명 (예: 'Wheat')
    element : str - 요소 (Production, Area harvested, Yield)
    
    Returns:
    --------
    DataFrame - 연도 × 국가 피벗 테이블
    """
    # 필터링
    mask = (df['Item'] == item_name) & (df['Element'] == element)
    df_filtered = df[mask].copy()
    
    if len(df_filtered) == 0:
        print(f"⚠️ '{item_name}' - '{element}' 데이터 없음")
        return None
    
    # Pivot
    df_pivot = df_filtered.pivot_table(
        index='Year',
        columns='Area',
        values='Value',
        aggfunc='sum'
    ).reset_index()
    
    # 전체 합계
    numeric_cols = [c for c in df_pivot.columns if c != 'Year']
    df_pivot['World_Total'] = df_pivot[numeric_cols].sum(axis=1)
    
    return df_pivot


def preprocess_temperature(df):
    """
    기온 변화 데이터 전처리
    - 연간 데이터만 추출
    - Wide format 변환
    """
    # 연간 데이터 필터
    df_annual = df[df['Months'] == 'Meteorological year'].copy()
    
    # Pivot
    df_pivot = df_annual.pivot_table(
        index='Year',
        columns='Area',
        values='Value',
        aggfunc='mean'
    ).reset_index()
    
    return df_pivot


def preprocess_korea_trade(df, element='Import Quantity'):
    """
    한국 수입 데이터 전처리
    """
    df_filtered = df[df['Element'] == element].copy()
    
    df_pivot = df_filtered.pivot_table(
        index='Year',
        columns='Item',
        values='Value',
        aggfunc='sum'
    ).reset_index()
    
    return df_pivot
```

```python
def create_integrated_dataset():
    """
    모든 재료의 생산량과 기온 데이터를 통합한 분석용 데이터셋 생성
    """
    # 기온 데이터 로드
    df_temp = pd.read_csv(PROCESSED_DIR / 'temperature_processed.csv')
    
    integrated_data = []
    
    items_files = {
        'Wheat': 'wheat.csv',
        'Tomatoes': 'tomatoes.csv',
        'Potatoes': 'potatoes.csv',
        'Lettuce': 'lettuce_and_chicory.csv',
        'Chillies': 'chillies_and_peppers_dry.csv',
        'Beef': 'beef.csv'
    }
    
    for item_name, filename in items_files.items():
        filepath = PROCESSED_DIR / filename
        if filepath.exists():
            df = pd.read_csv(filepath)
            
            for _, row in df.iterrows():
                integrated_data.append({
                    'Year': row['Year'],
                    'Item': item_name,
                    'World_Production_Tonnes': row.get('World_Total', np.nan),
                    'Temp_Anomaly_C': row.get('Temp_Anomaly', np.nan)
                })
    
    df_integrated = pd.DataFrame(integrated_data)
    
    # 연간 변화율 계산
    df_integrated['YoY_Change_Pct'] = df_integrated.groupby('Item')['World_Production_Tonnes'].pct_change() * 100
    
    return df_integrated


# 통합 데이터셋 생성
```

```python
def validate_data_quality(df, name):
    """
    데이터 품질 검증

    print(f"\n📋 {name} 데이터 품질 검증")
    print("-" * 40)
    
    # 기본 정보
    print(f"행 수: {len(df):,}")
    print(f"열 수: {len(df.columns)}")
    
    # 결측치
    missing = df.isnull().sum()
    if missing.sum() > 0:
        print(f"\n결측치:")
        for col, count in missing[missing > 0].items():
            print(f"   - {col}: {count} ({count/len(df)*100:.1f}%)")
    else:
        print("결측치: 없음")
    
    # 수치형 컬럼 통계
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    if len(numeric_cols) > 0:
        print(f"\n수치형 변수 요약:")
        print(df[numeric_cols].describe().round(2))
    
    return True
```
